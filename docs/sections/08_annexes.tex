\subsection{Éléments à déposer comme archives}

\begin{itemize}
    \item code source du dépôt ;
    \item PDF généré depuis ce dossier LaTeX ;
    \item fichiers de modélisation 3D : boîtier, support et pièce écran ;
    \item exports STL déjà présents dans \texttt{docs/assets/cad} ;
    \item schéma de câblage, idéalement au format image ou PDF ;
    \item photos du prototype ;
    \item éventuelles captures de l'interface web ;
    \item support de soutenance ;
    \item poster de présentation.
\end{itemize}
